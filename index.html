⚡ Zero-Touch Setup: How to Run This Automatically Every Day at 7:00 AM

Choose your preferred hands-free setup: A free Python cron script (GitHub Actions), or no-code (Make.com / n8n).

This lightweight script runs using the official Google GenAI SDK. It executes at 7:00 AM every morning via GitHub Actions (free) or your local laptop cron. It generates the text, verifies it, formats it, and automatically emails it into your Evernote inbox. 

import os
import smtplib
from email.mime.text import MIMEText
from email.mime.multipart import MIMEMultipart
from google import genai
from google.genai import types

# 1. Initialize Gemini Client
client = genai.Client()

PROMPT_DISCOVERY = """
Search and identify an inspiring Malaysian Bumiputera business (SME, startup, or established enterprise).
Write the success factors and challenges of their business strictly in bullet point form without any header or title.
End strictly with 6 to 10 suggested hashtags by AI.
"""

def generate_bumi_case_study():
    response = client.models.generate_content(
        model='gemini-2.5-flash',
        contents=PROMPT_DISCOVERY,
        config=types.GenerateContentConfig(
            tools=[types.Tool(google_search=types.GoogleSearch())],
            temperature=0.7,
        )
    )
    return response.text

def send_to_evernote(content, subject="[7AM Daily] Bumiputera Business Case Study @BumiBiz #CaseStudy"):
    evernote_email = os.environ.get("EVERNOTE_INBOX_EMAIL", "your_secret_address@m.evernote.com")
    smtp_user = os.environ.get("SMTP_USER")
    smtp_pass = os.environ.get("SMTP_PASS")
    
    msg = MIMEMultipart()
    msg['From'] = smtp_user
    msg['To'] = evernote_email
    msg['Subject'] = subject
    msg.attach(MIMEText(content, 'plain'))
    
    with smtplib.SMTP_SSL("smtp.gmail.com", 465) as server:
        server.login(smtp_user, smtp_pass)
        server.send_message(msg)
    print("Successfully sent to Evernote!")

if __name__ == "__main__":
    result = generate_bumi_case_study()
